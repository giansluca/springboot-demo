swagger: http://localhost:8080/swagger-ui/index.html

-- envs:
    APP_NAME=springboot-demo
    LOG_LEVEL=INFO
    ENV=local
    POSTGRES_HOST=localhost
    POSTGRES_PORT=5432
    POSTGRES_USER=docker
    POSTGRES_PASS=12345
    MONGO_HOST=localhost
    MONGO_PORT=27017
    MONGO_USER=docker
    MONGO_PASS=12345

-- local run -->
    ** network and DB:
    - docker network create -d bridge my-network
    - docker run --name postgres-1 --network=my-network -e POSTGRES_USER=docker -e POSTGRES_PASSWORD=12345 -e POSTGRES_DB=postgress-db-1 -d -p 5432:5432 postgres:18.3
    - docker run --name mongo-1 --network=my-network -e MONGO_INITDB_ROOT_USERNAME=docker -e MONGO_INITDB_ROOT_PASSWORD=12345 -e MONGO_INITDB_DATABASE=mongo-db-1 -d -p 27017:27017 mongo:8.0

    ** build and run application:
    - docker compose up -d
    OR
    - docker buildx build -t gianlucamori/springboot-demo:latest .
    - docker run --env-file docker.env --name springboot-demo-1 --network=my-network -d -p 8080:8080 gianlucamori/springboot-demo:latest java -jar /release/app/springboot-demo.jar

    ** tag and push to docker hub:
    - docker tag gianlucamori/springboot-demo:latest (if not tagged)
    - docker push gianlucamori/springboot-demo:latest

-- kubernetes on minikube -->
    ** run cluster locally with minikube:
    - minikube start
    - kubectl create namespace black-bull
    - kubectl apply -f k8s/

        ** if want to load images from local docker registry to minikube:
        - minikube image load mongo:8.0
        - minikube image load postgres:18.3
        - minikube image load gianlucamori/springboot-demo:latest
        - eval $(minikube docker-env)

    ** access pod from outside:
    - minikube service list
    - minikube service springboot-demo-service -n black-bull --url --> create a tunnel to access the service from localhost

      example url --> http://127.0.0.1:50617

-- install traefik as gateway api controller -->
    create a tunnel to access traefik dashboard and services from localhost
    - minikube tunnel

    ** create self-signed certificate for traefik:
    - openssl req -x509 -nodes -days 365 -newkey rsa:2048 \
        -keyout k8s/certificates/tls.key -out k8s/certificates/tls.crt \
        -subj "/CN=*.demo.localhost"

    ** create a secret in kubernetes with the certificate:
    - kubectl create secret tls local-selfsigned-tls \
        --cert=k8s/certificates/tls.crt \
        --key=k8s/certificates/tls.key \
        --namespace black-bull

    ** install traefik with helm:
    - helm repo add traefik https://traefik.github.io/charts
    - helm repo update
    - helm install traefik traefik/traefik \
        --version 39.0.4 \
        --values k8s/gateway-api/values.yaml \
        --namespace black-bull

    ** apply gateway api resources:
    - kubectl apply -f k8s/http-route/

    ** traefik dashboard:
    - https://gmkube.dashboard.localhost

    ** checks:
    curl -vLk http://gmkube-springboot-demo.localhost/api/v1/quote
        -v --> verbose output
        -L --> follow redirects
        -k --> allow insecure connections (ignore TLS certificate issues, for dev self-signed certificate)

    - kubectl get pods -n black-bull
    - kubectl get gatewayclasses
    - kubectl get gateway -n black-bull
    - kubectl get svc -n black-bull
    - kubectl get svc traefik -n black-bull
    - kubectl get httproute -n black-bull
